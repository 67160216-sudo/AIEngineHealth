# ใช้ Node.js LTS เป็น Base Image
FROM node:18-alpine

# กำหนด Working Directory
WORKDIR /app

# ก๊อปปี้ไฟล์ package.json และติดตั้ง Dependencies
COPY package*.json ./
RUN npm install --production

# ก๊อปปี้ Source Code ทั้งหมด
COPY . .

# เปิด Port 3000
EXPOSE 3000

# คำสั่งสำหรับเริ่มรันแอปพลิเคชัน
CMD ["node", "server.js"]
